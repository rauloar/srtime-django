import type { DailyAttendance } from '../types/contracts';

export const getStatusColor = (record: DailyAttendance): string => {
    // Priority: backend-provided color
    if (record.status_info?.color) {
        return record.status_info.color;
    }

    // Fallback: CSS variable mapping
    const map: Record<string, string> = {
        "Normal": "var(--att-normal)",
        "Absent": "var(--att-absent)",
        "Late": "var(--att-late)",
        "Early": "var(--att-early)",
        "Partial": "var(--att-partial)",
        "Rest Day": "var(--att-rest-day)",
    };

    return map[record.status] ?? "var(--text-secondary)";
};

const fetchData = async () => {
    setLoading(true);
    try {
        const res = await getDailyReports(
            startDate, 
            endDate, 
            departmentId ? parseInt(departmentId) : undefined,
            userIdFilter,
            nameFilter
        );
        // Validación defensiva: protege si backend cambia estructura
        if (!Array.isArray(res)) {
            console.error('Backend returned non-array response:', res);
            setData([]);
            return;
        }
        // Sort by Department then Date
        res.sort((a, b) => (a.employee?.department_name || '').localeCompare(b.employee?.department_name || '') || a.date.localeCompare(b.date));
        setData(res);
    } catch (e) {
        console.error(e);
    } finally {
        setLoading(false);
    }
};

PROTECCIÓN AGREGADA:
1. Valida que res sea array antes de .sort()
2. Si no es array:
   - Log error a consola con respuesta incorrecta
   - Setea data vacía (evita crashes)
   - Sale early (no ejecuta sort)
3. Si es array, continúa normal

ESCENARIOS PROTEGIDOS:
- Backend devuelve { error: "..." } en lugar de []
- Backend cambia a paginación sin avisar: { results: [], count: 0 }
- Backend envía null o undefined
- Typo en endpoint devuelve HTML 404

IMPACTO:
+ Previene crashes por .sort() en no-array
+ Log indica problema en backend
+ UI muestra loading→sin-datos en lugar de pantalla roja
