VI. IMPLEMENTACIÓN DE BREADCRUMB GLOBAL
================================================================================

Recomendación: Crear componente <Breadcrumb /> reutilizable

Estructura:
────────────────────────────────────────────────────────────────────────────

export interface BreadcrumbItem {
  label: string;
  path: string;
  icon?: ReactNode;
}

<Breadcrumb items={[
  { label: 'Dashboard', path: '/dashboard', icon: <Home /> },
  { label: 'Reports', path: '/attendance/reports' },
  { label: 'Juan Pérez', path: '/employees/5' },
  { label: '2026-02-10', path: '# ' (current) }
]} />

// Renderizado:
Dashboard > Reports > Juan Pérez > 2026-02-10


Uso en DayViewPage:
────────────────────────────────────────────────────────────────────────────

const breadcrumbs: BreadcrumbItem[] = [
  { label: 'Dashboard', path: '/dashboard' },
  { label: 'Reports', path: '/attendance/reports' },
  { label: data?.employee_name || '', path: '#' },
  { label: date, path: '#' }
];

<Breadcrumb items={breadcrumbs} />


Beneficio:
  ✓ Usuario siempre sabe dónde está
  ✓ Puede volver un paso atrás
  ✓ Coherencia visual
